import './AvatarJugador.css'
function AvatarJugador({ src, apodo, onCambiar, error }) {
  return <div className="avatar-jugador"><div className="avatar-retrato"><img className={!src || src === '/favicon.svg' ? 'avatar-predeterminado' : undefined} src={src || '/favicon.svg'} alt={'Avatar de ' + apodo} /></div><div className="avatar-control"><input id="avatar-archivo" className="avatar-archivo" type="file" accept="image/png,image/jpeg,image/webp" onChange={onCambiar} aria-describedby={error ? }
export default AvatarJugador
